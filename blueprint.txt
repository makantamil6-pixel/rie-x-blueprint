# Helper Function to parse the text asset file safely
def load_blueprint_text():
    content_dict = {}
    if os.path.exists("blueprint.txt"):
        with open("blueprint.txt", "r") as f:
            raw_text = f.read()
        blocks = raw_text.split("[")
        for block in blocks:
            if "]" in block:
                key, val = block.split("]", 1)
                content_dict[key.strip().lower()] = val.strip()
    return content_dict
